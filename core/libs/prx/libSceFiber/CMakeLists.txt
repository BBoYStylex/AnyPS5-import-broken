    target_include_directories(fiber_tests PRIVATE ${LIBS_INCLUDE_DIR})
    target_link_libraries(fiber_tests PRIVATE libSceFiber libkernel libc)
    configure_windows_unwind(fiber_tests)
    add_test(NAME fiber COMMAND fiber_tests)
    if(WIN32)
        add_test(NAME fiber_guard COMMAND fiber_tests guard)
        set_tests_properties(fiber_guard PROPERTIES ENVIRONMENT "APS5_FIBER_GUARD=1")
        set_tests_properties(fiber fiber_guard PROPERTIES ENVIRONMENT_MODIFICATION
            "PATH=path_list_prepend:$<TARGET_FILE_DIR:libSceFiber>")
    endif()
endif()
